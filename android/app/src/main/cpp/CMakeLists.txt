cmake_minimum_required(VERSION 3.22)
project(pet_inference LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Gradle's debug variant previously compiled the entire inference engine at O0.
# Keep debug symbols, but optimize CPU math in the debug APK used on phones.
add_compile_options("$<$<CONFIG:Debug>:-O3>")
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(LLAMA_SOURCE ${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native/vendor/pinned-llama)
add_subdirectory(${LLAMA_SOURCE} llama)
add_library(pet_inference SHARED pet_inference.cpp)
target_link_libraries(pet_inference PRIVATE llama android log)
target_include_directories(pet_inference PRIVATE ${LLAMA_SOURCE}/vendor)
